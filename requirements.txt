# sidecar dependencies.
#
# Python 3.8-3.14. SDK 2.0.16 declares python_requires='>=3.8,<3.15' with
# explicit cryptography/grpcio pins for 3.14, so the default python3 is fine:
#
#     python3 -m venv .venv && ./.venv/bin/pip install -r requirements.txt
#
# Unpinned on purpose: this is a personal tool on two machines, not a
# distributed package, and pinning here would mean maintaining it. If a
# transitive break ever shows up, pin the offender then, not pre-emptively.

# Web server
fastapi
uvicorn

# Webull broker API — reads AND the order path (see rule 3 in CLAUDE.md).
# Pulls in paho-mqtt and grpcio for the streaming clients, which stream.py does
# use; both need build tools on some Pythons.
webull-openapi-python-sdk

# TraderDaddy Pro JSON-RPC, Telegram, and ntfy all go over plain HTTP
requests

# Claude chat panel. Also needs the `claude` CLI binary, which this package does
# NOT ship (only the TypeScript SDK does):
#     npm i -g @anthropic-ai/claude-code
# Without it, chat fails at runtime rather than at install.
claude-agent-sdk

# Claude Desktop MCP server (mcp_server.py). Only needed on the machine running
# Claude Desktop, which may not be the machine running sidecar. httpx is the
# HTTP client it uses to reach sidecar.
#
# Pinned to >=2 because 2.0 renamed FastMCP to MCPServer and dropped
# mcp.server.fastmcp. mcp_server.py targets the 2.x API, so an older install
# fails at import — this is the one dependency where the floor is load-bearing.
mcp>=2
httpx

# Optional. Powers `notify.py --qr`, which puts the ntfy subscribe URL on screen
# as a QR so the topic never has to be typed, pasted into a chat, or read off a
# stream. Pure python, no compiler. Without it --qr prints the URL instead.
qrcode

# Vesper LangGraph Quant Agent & Outcome Memory
langgraph
pydantic>=2.0
python-dotenv>=1.0
typing_extensions>=4.0
chromadb>=0.5

# vesper/graph.py: persistent (disk-backed) checkpointer for human-approval
# threads. MemorySaver (langgraph's built-in default) loses every paused
# thread on a process restart -- fine for a single watched `vesper scan` run,
# a real gap now that `vesper loop` is a long-lived daemon that can crash or
# restart mid-approval-wait.
langgraph-checkpoint-sqlite
aiosqlite

# vesper/bot/inbound.py: the aiohttp webhook server and Discord Ed25519
# signature verification (cryptography's Ed25519PublicKey).
aiohttp
cryptography

# Discord gateway client with dynamic interactive approval buttons
discord.py
